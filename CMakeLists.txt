cmake_minimum_required(VERSION 3.6)

option(VITASDK_BUILD "set to ON if building for vita" OFF)
option(FLATPAK "set to ON if building a flatpak" OFF)
option(VSP_NETWORKING "enable SDL_net functionality" ON)
option(VSP_USE_LIBLCF "enable liblcf functionality" ON)
option(VSP_USE_LIBAVIF "enable libavif functionality" ON)
option(USE_FMT_FORMAT "Use fmt::format instead of std::format" OFF)

option(VSP_VENDORED_ALL_OFF "turn this on to disable all vendored libraries" OFF)
option(VSP_VENDORED_SDL "use vendored SDL" ON)
option(VSP_VENDORED_SDLIMAGE "use vendored SDL_image" ON)
option(VSP_VENDORED_SDLTTF "use vendored SDL_ttf" ON)

if(VITASDK_BUILD)
    if(DEFINED ENV{VITASDK})
        set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")
    else()
        message(FATAL_ERROR "Please define VITASDK to point to your SDK path!")
    endif()
    include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)

#platform-specific setup
if (EMSCRIPTEN)
    set(CONFIG_MULTITHREAD OFF)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -msimd128")
    set(COMPILE_C_ONLY ON)
    set(BUILD_SHARED_LIBS OFF)

    set(VSP_NETWORKING OFF)
    set(VSP_USE_LIBLCF OFF)
endif()
if (WIN32)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static-libstdc++ -Wno-unused-command-line-argument")

    set(SDLIMAGE_AVIF OFF)
    set(SDLIMAGE_VENDORED OFF)

#also libavif is already included there by default for some reason


#add vendored libs
if (VSP_VENDORED_SDL)
    set(SDL_TESTS OFF)
    set(SDL_TEST_LIBRARY OFF)
    set(SDL_EXAMPLES OFF)
    add_subdirectory(external/SDL)
endif()
if (VSP_VENDORED_SDLIMAGE)
    add_subdirectory(external/SDL_image)
endif()
if (VSP_VENDORED_SDLTTF)
    add_subdirectory(external/SDL_ttf)
endif()

if (VSP_NETWORKING AND VSP_VENDORED_SDLNET)
    add_subdirectory(external/SDL_net)
endif()

if (VSP_USE_LIBLCF AND VSP_VENDORED_LCF)
    set(LIBLCF_WITH_INI OFF)
    set(LIBLCF_WITH_ICU OFF)
    set(LIBLCF_WITH_XML OFF)
    set(LIBLCF_MAIN_PROJECT OFF)
    add_subdirectory(external/liblcf)
endif()

if (VSP_USE_LIBAVIF AND VSP_VENDORED_AVIF)
    set(AVIF_LIBYUV "LOCAL")
    set(AVIF_CODEC_AOM OFF)
    set(AVIF_CODEC_LIBGAV1 "LOCAL")
    set(AVIF_CODEC_SVT "LOCAL")
    add_subdirectory(external/libavif)
endif()

#these are static and can be used on virtually any platform
if (VSP_VENDORED_ZLIB)
    set(ZLIB_BUILD_TESTING OFF)
    set(ZLIB_BUILD_SHARED OFF)
    add_subdirectory(external/zlib)
endif()
if (USE_FMT_FORMAT AND VSP_VENDORED_FMT)
    add_subdirectory(external/fmt)
endif()
